using PhyloNetworks
using Distributions
using Random

# PhyloNetworks renamed several functions in v0.16+ (readTopology -> readnewick, etc).
# These small wrappers make the script work with either version.
_readnet(s)  = isdefined(PhyloNetworks, :readnewick)  ? readnewick(s)  : readTopology(s)
_writenet(n) = isdefined(PhyloNetworks, :writenewick) ? writenewick(n; round=false) : writeTopology(n, round=false)
_getchild(e) = isdefined(PhyloNetworks, :getchild)     ? getchild(e)   : getChild(e)

const ATOL = 1e-8   # tolerance for "exactly 0.4" / "exactly 0.3" matches

"""
    process_network(newick::AbstractString) -> String

Read a network, resample branch lengths that equal 0.4 (Uniform(0.2,0.4))
and hybrid gammas that equal 0.3 (Uniform(0.25,0.3), partner = 1 - value),
then return the modified Newick string.
"""
function process_network(newick::AbstractString)
    net = _readnet(String(newick))

    ## ---- 1) branch lengths exactly 0.4 ----
    for e in net.edge
        if e.length !== missing && !isnan(e.length) && isapprox(e.length, 0.4; atol=ATOL)
            e.length = rand(Uniform(0.2, 0.4))
        end
    end

    ## ---- 2) hybrid gammas exactly 0.3, keep pair summing to 1 ----
    hybedges = [e for e in net.edge if e.hybrid]
    groups = Dict{Int, Vector{PhyloNetworks.Edge}}()
    for e in hybedges
        childnode = _getchild(e)
        push!(get!(groups, childnode.number, PhyloNetworks.Edge[]), e)
    end

    for (_, es) in groups
        length(es) == 2 || continue   # normal case: exactly 2 parent edges per hybrid node
        g1, g2 = es[1].gamma, es[2].gamma
        if (g1 !== missing && isapprox(g1, 0.3; atol=ATOL)) ||
           (g2 !== missing && isapprox(g2, 0.3; atol=ATOL))
            newval  = rand(Uniform(0.25, 0.3))
            compval = 1 - newval
            es[1].gamma = isapprox(g1, 0.3; atol=ATOL) ? newval : compval
            es[2].gamma = isapprox(g2, 0.3; atol=ATOL) ? newval : compval
        end
    end

    return _writenet(net)
end

"""
    process_file(infile, outfile)

Read one network per line, transform each, write results (one per line).
"""
function process_file(infile::AbstractString, outfile::AbstractString)
    open(outfile, "w") do io
        for line in eachline(infile)
            isempty(strip(line)) && continue
            println(io, process_network(line))
        end
    end
end
